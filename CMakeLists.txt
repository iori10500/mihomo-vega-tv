cmake_minimum_required(VERSION 3.19)
project(mihomotv LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(turbomoduleAPI CONFIG REQUIRED)
find_package(Apmf REQUIRED)
find_package(Kepler REQUIRED)
find_package(KeplerNetworkManagerInterface REQUIRED)
kepler_add_idl_import_library(vpn_idl PACKAGES com.amazon.kepler.net.vpn.vpnmanager com.amazon.kepler.net.vpn.wireguard)
kepler_add_turbo_module_library(mihomotv kepler/AutoLinkInit.cpp kepler/Core.cpp kepler/Vpn.cpp kepler/generated/CoreSpec.cpp)
target_link_libraries(mihomotv PRIVATE dl vpn_idl)
target_compile_options(mihomotv PRIVATE "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=.")
install(CODE "if(NOT EXISTS \"${CMAKE_CURRENT_SOURCE_DIR}/native/libtvcore.so\" AND NOT DEFINED ENV{MIHOMO_BOOTSTRAP})
  message(FATAL_ERROR \"Build the Mihomo core with npm run build\")
endif()")
install(FILES native/libtvcore.so DESTINATION lib OPTIONAL)
install(TARGETS mihomotv)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.measure-memory")
  target_compile_definitions(mihomotv PRIVATE MIHOMO_MEASURE_MEMORY)
endif()
